def review_figure(t, q_kpa, h_m, limit_kpa):
    """Two stacked panels, q with its limit on top and altitude below; returns (fig, ax_q, ax_h)."""

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np


def line_peaks(ax):
    """Label -> largest finite y value, for every labeled line with data on ax."""
    peaks = {}
    for line in ax.lines:
        label = line.get_label()
        y = np.asarray(line.get_ydata(), dtype=float)
        if label.startswith("_") or not np.any(~np.isnan(y)):
            continue
        peaks[label] = float(np.nanmax(y))
    return peaks
--- hint
`ax.lines` holds every line on the Axes; each has `get_label()` and `get_ydata()`.
--- hint

import os

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np


def save_cases(cases, folder, limit_kpa):
    """One PNG per case in folder, each figure closed after saving; returns [(path, title), ...]."""
    saved = []
    for k, (t, q_kpa) in enumerate(cases):
        peak = float(np.max(q_kpa))
        title = f"Case {k:03d}: peak {peak:.1f} kPa, margin {limit_kpa - peak:.1f} kPa"
        fig, ax = plt.subplots(figsize=(6.4, 3.6))
        ax.plot(t, q_kpa)
        ax.axhline(limit_kpa, color="red", linestyle="--")
        ax.set_title(title)
        path = os.path.join(folder, f"case_{k:03d}.png")
        fig.savefig(path, dpi=100)
        plt.close(fig)
        saved.append((path, title))
    return saved

+++ question | Where the unnamed calls went
--- ask
What does this print?

```python
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt

fig, axs = plt.subplots(2, 2)
axs[0, 0].plot([0, 1], [0, 1])
plt.plot([0, 1], [1, 0])
fig2, ax = plt.subplots()
plt.plot([0, 1], [2, 2])
print([len(a.lines) for a in axs.flat], len(ax.lines))
```
--- answer
[1, 0, 0, 1] 1
[1, 0, 0, 1], 1
[1,0,0,1] 1
--- why
`plt.subplots(2, 2)` makes four Axes, and the last one made, the bottom right `axs[1, 1]`, becomes the current axes. Drawing on `axs[0, 0]` by name does not move that pointer, so the first `plt.plot` lands on `axs[1, 1]`. Making `fig2` moves the pointer again, to its only Axes, so the second `plt.plot` lands there. `axs.flat` walks the grid row by row: top left, top right, bottom left, bottom right, giving `[1, 0, 0, 1]`.

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np


def stack_panels(labels, panel_height_in):
    """One shared-x panel per label, always returned as a 1-D array of Axes."""
    n = len(labels)
    fig, axs = plt.subplots(n, 1, sharex=True, squeeze=False,
                            figsize=(7.0, panel_height_in * n), layout="constrained")
    axs = axs[:, 0]
    for ax, label in zip(axs, labels):
        ax.set_ylabel(label)
    axs[-1].set_xlabel("Time since liftoff (s)")
    return fig, axs
--- hint

def track_layout():
    """Tall track on the left, three shared-x time panels on the right."""
    fig = plt.figure(figsize=(9.0, 5.0), layout="constrained")
    gs = fig.add_gridspec(3, 2, width_ratios=[3, 2])

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np


def altitude_mach_panel(ax, t, h_m, mach):
    """Altitude on ax, Mach on a twin with its own y axis; one legend on ax. Returns the twin."""
    twin = ax.twinx()
    line_h, = ax.plot(t, h_m / 1000.0, color="C0", label="Altitude")
    line_m, = twin.plot(t, mach, color="C1", linestyle="--", label="Mach")
    ax.set_xlabel("Time since liftoff (s)")
    ax.set_ylabel("Altitude (km)")
    twin.set_ylabel("Mach (-)")
    ax.legend(handles=[line_h, line_m])

+++ practice | Where two curves cross on a twin-axis panel
--- task
On a twin-axis panel, curve 1 is drawn against the left axis with limits `lim1 = (bottom, top)`, and curve 2 against the right axis with limits `lim2`. Where the two curves appear to cross on the page depends only on those limits. Write `page_crossings(t, y1, lim1, y2, lim2)` that returns the times where they cross on the page.

- A value `y` drawn on an axis with limits `(bottom, top)` sits at page height `(y - bottom) / (top - bottom)`: 0 at the bottom edge, 1 at the top.
- Let `d` be the page height of curve 1 minus that of curve 2, at each sample. Count `d` equal to 0 as positive.

- Use NumPy on whole arrays: no `for` or `while` loop anywhere, not even in a comprehension.

--- solution
import numpy as np


def page_crossings(t, y1, lim1, y2, lim2):
    """Times where two curves on different axis limits cross on the page."""
    t = np.asarray(t, dtype=float)
    h1 = (np.asarray(y1, dtype=float) - lim1[0]) / (lim1[1] - lim1[0])
    h2 = (np.asarray(y2, dtype=float) - lim2[0]) / (lim2[1] - lim2[0])
    d = h1 - h2
    up = d >= 0
    k = np.nonzero(up[:-1] != up[1:])[0]
    t_cross = t[k] + (t[k + 1] - t[k]) * d[k] / (d[k] - d[k + 1])
    return np.round(t_cross, 1).tolist()

--- hint

def ensemble_panel(ax, t, runs, limit):
    """Mean ± 3 sigma band, mean, limit, and a tint where the band tops the limit."""

import numpy as np


def band(runs, how):
    """Band edges across runs at each time, ignoring NaN; NaN where fewer than 2 values."""
    runs = np.asarray(runs, dtype=float)
    count = np.sum(~np.isnan(runs), axis=0)
    if how == "sigma":
        mean = np.nanmean(runs, axis=0)

    else:
        raise ValueError(f"unknown band: {how!r}")
    enough = count >= 2
    return np.where(enough, lo, np.nan), np.where(enough, hi, np.nan)

--- solution
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np


def filter_check(ax, t, err, P, i):
    """Plot err inside the filter's own ±3 sigma band; return (fraction inside, verdict)."""
    sigma = np.sqrt(P[:, i, i])
    ax.fill_between(t, -3.0 * sigma, 3.0 * sigma, alpha=0.25, label="filter ±3σ")
    ax.plot(t, err, linewidth=0.5, label="true error")
    inside = float(np.mean(np.abs(err) <= 3.0 * sigma))
    ratio = float(np.max(np.abs(err) / sigma))
    if inside < 0.99:
        verdict = "optimistic"
    elif ratio < 1.5:
        verdict = "pessimistic"
    else:
        verdict = "consistent"
    return round(inside, 4), verdict

def log_or_symlog(ax, t, y):
    """Plot y on the scale that can show it: log, symlog, or linear; returns (scale, linthresh)."""
    ax.plot(t, y)
    values = np.asarray(y, dtype=float)
    values = values[~np.isnan(values)]
    nonzero = np.abs(values[values != 0])
    if values.size and np.all(values > 0):
        ax.set_yscale("log")
        return "log", None

        return "symlog", float(linthresh)
    return "linear", None

import matplotlib.pyplot as plt
import numpy as np


def root_locus(ax, num, den, gains):
    """Plot closed-loop roots as dots; return (K, w) where they first reach the imaginary axis."""
    num = np.asarray(num, dtype=float)
    den = np.asarray(den, dtype=float)
    padded = np.concatenate([np.zeros(den.size - num.size), num])
    roots = np.array([np.roots(den + k * padded) for k in gains])
    ax.plot(roots.real.ravel(), roots.imag.ravel(), ".", markersize=2)
    ax.axvline(0.0, color="0.4")
    ax.set_aspect("equal")
    reached = roots.real.max(axis=1) >= -1e-9
    if not reached.any():
        return None
    j = int(np.argmax(reached))
    return float(gains[j]), round(float(np.abs(roots[j].imag).max()), 2)

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np


def one_legend(fig):
    """One figure legend above the panels, each label once; returns the labels."""
    handles, labels = [], []
    for ax in fig.axes:
        for handle, label in zip(*ax.get_legend_handles_labels()):
            if label not in labels:
                handles.append(handle)
                labels.append(label)
    if not labels:
        return []
    for ax in fig.axes:
        if ax.get_legend() is not None:
            ax.get_legend().remove()
    fig.legend(handles, labels, loc="outside upper center", ncols=len(labels))
    return labels

+++ question | Run numbers that come out twice
--- ask
A plot has Monte Carlo run numbers 0 to 5 on its x axis. The automatic ticks land at 0, 0.5, 1, 1.5, … and someone adds `ax.xaxis.set_major_formatter(StrMethodFormatter("{x:.0f}"))` to hide the fractions. Now the labels read 0, 0, 1, 2, 2, 2, 3, 4, 4, 4, 5. What is the right fix?
--- choice
Use `"{x:d}"` instead, so the formatter writes whole numbers.
--- choice correct
Change the locator, not the formatter: `ax.xaxis.set_major_locator(MaxNLocator(integer=True))`, so ticks only sit on whole numbers.
--- choice
Set `AutoMinorLocator(2)`, so the half-way ticks become unlabeled minor ticks.
--- choice
Call `ax.set_xticklabels(range(6))` to overwrite the labels with 0 to 5.
--- why
A formatter only decides what each tick **says**; the ticks at 0.5, 1.5 and so on are still there, and rounding their labels makes several ticks claim the same run, some of them at the wrong place. Where ticks go is the locator's job, and `MaxNLocator(integer=True)` allows only whole-number positions. `"{x:d}"` fails on a float, `set_xticklabels` pastes six labels onto however many ticks exist, and minor ticks do not remove the major ticks at the fractions.

def grey_gap(c1, c2):
    """Greyscale values (0 to 255) of two colors and the gap between them."""

import numpy as np
from matplotlib.colors import Normalize, TwoSlopeNorm


def diverging_norm(data, center):
    """A norm that maps center to 0.5 for any data, NaN ignored."""
    values = np.asarray(data, dtype=float)
    values = values[~np.isnan(values)]
    if values.size and values.min() < center < values.max():
        return TwoSlopeNorm(vmin=float(values.min()), vcenter=center, vmax=float(values.max()))
    d = float(np.max(np.abs(values - center))) if values.size else 0.0
    if d > 0:
        return Normalize(center - d, center + d)
    return Normalize(center - 1.0, center + 1.0)

def small_text(fig, shown_width_in, min_pt):

VERSIONS = {"report": ((3.5, 2.4), 8, 300), "slide": ((12.0, 5.5), 18, 200)}


def two_versions(plot_fn, data):
    """Report and slide figures from one plotting function; returns pixel sizes and label size."""
    out = {}
    for name, (size, font_pt, dpi) in VERSIONS.items():

def playback(rate_hz, duration_s, speedup, fps):
    """Samples per frame, number of frames and movie length for a telemetry animation."""
    if (rate_hz * speedup) % fps != 0:

def padded(values):
    """Limits 5% of the span beyond the data on each side."""
    lo, hi = float(np.min(values)), float(np.max(values))
    pad = 0.05 * (hi - lo)
    return lo - pad, hi + pad


def descent_animation(t, x, h, step, fps):
    """Figure, animation, update function and frame list for a lander descent movie."""
    fig, ax = plt.subplots(figsize=(4, 3), layout="constrained")
    ax.set_xlim(*padded(x))
    ax.set_ylim(*padded(h))
    ax.set_xlabel("Crossrange (m)")
    ax.set_ylabel("Altitude (m)")
    trail, = ax.plot([], [], color="C0")
    dot, = ax.plot([], [], "o", color="C3")
    label = ax.text(0.02, 0.95, "", transform=ax.transAxes, va="top")

    def update(i):
        trail.set_data(x[: i + 1], h[: i + 1])
        dot.set_data([x[i]], [h[i]])
        label.set_text(f"t = {t[i]:.1f} s")
        return trail, dot, label

import numpy as np


def window_stats(t, x, lo, hi):
    """Number of real samples with lo <= t <= hi (limits in any order) and their peak |x|."""
    lo, hi = min(lo, hi), max(lo, hi)
    inside = (t >= lo) & (t <= hi) & ~np.isnan(x)
    count = int(inside.sum())
    if count == 0:

    return count, round(float(np.abs(x[inside]).max()), 3)

def live_stats(ax, t, x):
    """A corner text on ax showing the count and peak |x| in view, refreshed on every x-limit change."""

import io

import numpy as np
import pandas as pd


def imu_from_csv(text, liftoff):
    """IMU table in time order, one row per time, UTC index; also the sample rate in Hz."""
    raw = pd.read_csv(io.StringIO(text))
    met = raw["met_s"].to_numpy()
    order = np.argsort(met, kind="stable")
    met = met[order]
    keep = np.concatenate([[True], np.diff(met) > 0])
    table = raw.iloc[order[keep]].drop(columns="met_s")
    table.index = pd.Timestamp(liftoff, tz="UTC") + pd.to_timedelta(met[keep], unit="s")
    table.index.name = "time_utc"
    rate_hz = int(round(1.0 / np.median(np.diff(met[keep]))))
    return table, rate_hz

import numpy as np
import pandas as pd


def fill_short_gaps(s, max_rows):
    """Time-interpolate NaN runs of at most max_rows with real values on both sides; flag them."""
    missing = s.isna().to_numpy()
    gap_id = np.cumsum(~missing)                     # real values seen so far
    lengths = np.bincount(gap_id[missing], minlength=gap_id[-1] + 1)

    fill = missing & between & (lengths[gap_id] <= max_rows)
    value = s.where(~fill, s.interpolate(method="time"))
    return pd.DataFrame({"value": value, "filled": fill}, index=s.index)
--- hint
Number the gaps: the running count of real values is the same for every NaN in one gap. From that you can get each gap's length and whether real values lie on both sides of it.
--- hint

def phase_report(tel, rate_hz):
    """Per-phase sample count, peaks and duration (first-seen order), and the short one-second bins."""
    work = tel.copy()

--- ask

--- answer

5 and 1000
--- why
`resample` groups samples into time bins and returns one row per bin: 10 seconds in 2-second bins is 5 rows, each the mean of 200 samples. `rolling` keeps the index and computes a statistic over a window that slides along it, so it returns one value for every one of the 1000 samples. Downsampling is resample; a moving average is rolling.

import numpy as np
import pandas as pd


def compare_velocity(imu, gps, rate_hz):
    """GPS minus IMU velocity at each fix, nearest IMU sample within 0.6 sample spacings."""
    both = pd.merge_asof(gps, imu, on="t", direction="nearest", tolerance=0.6 / rate_hz,
                         suffixes=("_gps", "_imu"))
    residual = both["v_mps_gps"] - both["v_mps_imu"]
    matched = residual.notna()
    report = {"n_matched": int(matched.sum()), "n_missing": int((~matched).sum()),
              "rms": None, "worst_t": None}

        report["rms"] = round(float(np.sqrt((residual[matched] ** 2).mean())), 3)
        report["worst_t"] = float(both.loc[residual.abs().idxmax(), "t"])
    return report

`direction="forward"` answers "what was the first thing at or after this moment?": 1.0 gets the row at 2.0, 2.5 gets the row at 3.0, and 4.0 has nothing after it, so NaN. Backward would give `[NaN, 20.0, 30.0]`. Nearest would give `[20.0, 20.0, 30.0]`: 2.5 is equally far from 2.0 and 3.0, and on a tie pandas takes the earlier row. merge_asof never needs equal keys; that is what it is for.

import numpy as np


def minmax_decimate(t, x, n_bins):
    """Min and max of each chunk in time order, NaN ignored, a repeated sample kept once."""
    t = np.asarray(t, dtype=float)
    x = np.asarray(x, dtype=float)
    real = ~np.isnan(x)
    if n_bins >= x.size:
        return t[real], x[real]
    keep = []
    for chunk in np.array_split(np.arange(x.size), n_bins):
        chunk = chunk[real[chunk]]
        if chunk.size == 0:
            continue
        lo = chunk[np.argmin(x[chunk])]
        hi = chunk[np.argmax(x[chunk])]
        keep.extend(sorted({int(lo), int(hi)}))
    keep = np.array(keep, dtype=int)
    return t[keep], x[keep]

import numpy as np


def zoom_view(t, x, lo, hi, n_cols):
    """Min/max view of lo <= t <= hi (plus one sample each side) in n_cols bins, from full data."""
    start = max(int(np.searchsorted(t, lo, side="left")) - 1, 0)
    stop = min(int(np.searchsorted(t, hi, side="right")) + 1, t.size)
    tw, xw = t[start:stop], x[start:stop]
    if xw.size <= 2 * n_cols:
        return tw.copy(), xw.copy()
    keep = {0, xw.size - 1}
    for chunk in np.array_split(np.arange(xw.size), n_cols):
        keep.update((int(chunk[np.argmin(xw[chunk])]), int(chunk[np.argmax(xw[chunk])])))
    keep = sorted(keep)
    return tw[keep], xw[keep]

def flight_review(t, channels, events, band_on, runs):
    """Stacked shared-x panels, a 3-sigma band on one, event lines on all, one figure legend."""
    n = len(channels)
    fig, axs = plt.subplots(n, 1, sharex=True, squeeze=False, figsize=(7.0, 1.6 * n),

    for ax, (label, data) in zip(axs, channels):
        if label == band_on:
            mean = runs.mean(axis=0)
            sd = runs.std(axis=0, ddof=1)
            handles["mean ± 3σ"] = ax.fill_between(t, mean - 3.0 * sd, mean + 3.0 * sd,
                                                   color="C0", alpha=0.25)

    fig.legend([handles[name] for name in labels], labels, loc="outside upper center", ncols=len(labels))
    return fig, axs

- Evaluate the loop at `w = np.logspace(-2, 3, 20001)` rad/s, with `s = 1j * w`. Magnitude in dB is 20 log10 of the size; the phase in degrees must be continuous (unwrapped).
- **Gain crossover:** the first sample `k` where the magnitude is at least 0 dB and the next sample is below 0 dB. Interpolate between samples `k` and `k + 1` in log frequency: with `f = mag[k] / (mag[k] - mag[k+1])`, the crossover is at `log10(w) = log10(w[k]) + f * (log10(w[k+1]) - log10(w[k]))`, and the phase there is `phase[k] + f * (phase[k+1] - phase[k])`. `"w_c"` is that frequency rounded to 2 decimals; `"pm_deg"` is 180 plus that phase, rounded to 1 decimal.
- **Phase crossover:** the first sample `k` where the phase is above -180 and the next is at or below -180. Interpolate the same way with `f = (phase[k] + 180) / (phase[k] - phase[k+1])`. `"w_180"` is that frequency rounded to 2 decimals; `"gm_db"` is minus the interpolated magnitude there, in dB, rounded to 2 decimals.
- A crossing that never happens gives `None` for both of its entries.

import numpy as np


def interpolate_crossing(k, f, log_w, values):
    """Frequency and value a fraction f of the way from sample k to k + 1, in log frequency."""
    w = 10.0 ** (log_w[k] + f * (log_w[k + 1] - log_w[k]))
    return float(w), float(values[k] + f * (values[k + 1] - values[k]))


def loop_margins(num, den):
    """Gain crossover and phase margin, phase crossover and gain margin, interpolated."""
    w = np.logspace(-2, 3, 20001)
    s = 1j * w
    L = np.polyval(num, s) / np.polyval(den, s)
    mag = 20.0 * np.log10(np.abs(L))
    phase = np.degrees(np.unwrap(np.angle(L)))
    log_w = np.log10(w)
    out = {"w_c": None, "pm_deg": None, "w_180": None, "gm_db": None}
    k = np.nonzero((mag[:-1] >= 0.0) & (mag[1:] < 0.0))[0]
    if k.size:
        f = mag[k[0]] / (mag[k[0]] - mag[k[0] + 1])
        w_c, phase_c = interpolate_crossing(k[0], f, log_w, phase)
        out["w_c"], out["pm_deg"] = round(w_c, 2), round(180.0 + phase_c, 1)
    k = np.nonzero((phase[:-1] > -180.0) & (phase[1:] <= -180.0))[0]
    if k.size:
        f = (phase[k[0]] + 180.0) / (phase[k[0]] - phase[k[0] + 1])
        w_180, mag_180 = interpolate_crossing(k[0], f, log_w, mag)
        out["w_180"], out["gm_db"] = round(w_180, 2), round(-mag_180, 2)
    return out

def dispersion_report(t, runs, stop):
    """Where the 3-sigma band passes a hard stop, and how the runs really pile up against it."""
    mean = runs.mean(axis=0)
    sd = runs.std(axis=0, ddof=1)
    over = mean + 3.0 * sd > stop
    at_stop = (runs >= stop).mean(axis=0)
    top = np.percentile(runs, 99.865, axis=0)

        "worst_frac": round(float(at_stop.max()), 3),
        "top_max": round(float(top.max()), 3),
    }

def load_and_bin(text, width):

import io

import numpy as np
import pandas as pd


def align_imu_gps(imu_text, gps_text):
    """GPS fixes joined to the nearest IMU sample within 0.6 IMU spacings; returns (table, tol_ms)."""
    imu = pd.read_csv(io.StringIO(imu_text), parse_dates=["time_utc"])
    gps = pd.read_csv(io.StringIO(gps_text), parse_dates=["time_utc"])
    imu["time_utc"] = imu["time_utc"].astype("datetime64[ns, UTC]")
    gps["time_utc"] = gps["time_utc"].astype("datetime64[ns, UTC]")
    tol = 0.6 * imu["time_utc"].diff().median()
    right = imu.copy()
    right["t_imu"] = right["time_utc"]
    table = pd.merge_asof(gps, right, on="time_utc", direction="nearest", tolerance=tol)
    table["skew_ms"] = (table["t_imu"] - table["time_utc"]).dt.total_seconds() * 1000.0
    return table.drop(columns="t_imu"), round(tol.total_seconds() * 1000.0, 3)

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd


def mc_study(ax, samples, inputs, limit):
    """Peak q per run joined to its inputs, a wind-versus-peak scatter, and their correlation."""
    peaks = samples.groupby("run").agg(peak_kPa=("q_kPa", "max")).reset_index()
    table = pd.merge(inputs, peaks, on="run", how="left", validate="one_to_one")
    table["over"] = table["peak_kPa"] > limit
    has_peak = table["peak_kPa"].notna()
    wind, peak = table.loc[has_peak, "wind_ms"], table.loc[has_peak, "peak_kPa"]
    ax.scatter(wind, peak)
    ax.axhline(limit, color="black", linestyle="--")
    ax.set_xlabel("Wind speed (m/s)")
    ax.set_ylabel("Peak dynamic pressure (kPa)")
    return table, round(float(np.corrcoef(wind, peak)[0, 1]), 3)

--- solution
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np


class ZoomDecimator:
    """One line on ax that is min/max-decimated again from the full data on every x-limit change."""

    def __init__(self, ax, t, x, n_cols):
        self.t, self.x, self.n_cols = t, x, n_cols
        self.line, = ax.plot([], [], linewidth=0.8)
        ax.set_xlim(t[0], t[-1])
        ax.set_ylim(float(np.min(x)), float(np.max(x)))
        self.redraw(ax)
        ax.callbacks.connect("xlim_changed", self.redraw)

    def view(self, lo, hi):
        start = max(int(np.searchsorted(self.t, lo, side="left")) - 1, 0)
        stop = min(int(np.searchsorted(self.t, hi, side="right")) + 1, self.t.size)
        tw, xw = self.t[start:stop], self.x[start:stop]
        if xw.size <= 2 * self.n_cols:
            return tw, xw
        keep = {0, xw.size - 1}
        for chunk in np.array_split(np.arange(xw.size), self.n_cols):
            keep.update((int(chunk[np.argmin(xw[chunk])]), int(chunk[np.argmax(xw[chunk])])))
        keep = sorted(keep)
        return tw[keep], xw[keep]

    def redraw(self, ax):
        lo, hi = sorted(ax.get_xlim())
        self.line.set_data(*self.view(lo, hi))
